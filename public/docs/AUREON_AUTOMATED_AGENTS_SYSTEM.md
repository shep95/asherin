# PART 1: CORE CONCEPT

## 🤖 WHAT ARE AUTOMATED AGENTS?

**Simple Explanation:**
You tell Aureon: *"Send me an email every morning at 7am with my daily astrology prediction"*

Aureon creates an AI agent that:
- ✅ Runs automatically every day at 7am
- ✅ Generates your prediction using your birth chart
- ✅ Sends you an email
- ✅ Continues forever until you stop it

**The Power:**
Instead of asking Aureon to do things manually, create agents that do them automatically, forever.

```typescript
interface AutomatedAgent {
  id: string,
  name: string,
  description: string,
  
  // Trigger
  trigger: {
    type: 'schedule' | 'event' | 'webhook' | 'manual',
    
    // Schedule trigger
    schedule?: {
      frequency: 'once' | 'hourly' | 'daily' | 'weekly' | 'monthly' | 'custom',
      time?: string,  // "07:00"
      dayOfWeek?: number,  // 0-6
      dayOfMonth?: number,  // 1-31
      timezone: string,
      cron?: string  // Advanced: "0 7 * * *"
    },
    
    // Event trigger
    event?: {
      source: string,  // 'user_signup', 'payment_received', etc.
      filters?: Record<string, any>
    },
    
    // Webhook trigger
    webhook?: {
      url: string,
      secret: string
    }
  },
  
  // Actions (what the agent does)
  actions: Action[],
  
  // Output (where results go)
  output: {
    type: 'email' | 'sms' | 'slack' | 'webhook' | 'database' | 'file',
    config: Record<string, any>
  },
  
  // Status
  status: 'active' | 'paused' | 'deleted',
  
  // Stats
  totalRuns: number,
  successfulRuns: number,
  failedRuns: number,
  lastRun?: Date,
  nextRun?: Date,
  
  // Settings
  settings: {
    retryOnFailure: boolean,
    maxRetries: number,
    notifyOnFailure: boolean,
    timeout: number  // seconds
  },
  
  createdAt: Date,
  updatedAt: Date
}

interface Action {
  type: string,  // 'generate_prediction', 'send_email', 'analyze_video', etc.
  config: Record<string, any>,
  order: number
}
```

```typescript
// Agent Executor
async function executeAgent(agent: AutomatedAgent) {
  const executionId = crypto.randomUUID();
  const startTime = Date.now();
  
  try {
    // 1. Log execution start
    await logExecution(executionId, 'started', { agentId: agent.id });
    
    // 2. Execute each action in order
    let actionResults = [];
    
    for (const action of agent.actions.sort((a, b) => a.order - b.order)) {
      const actionResult = await executeAction(action, actionResults);
      actionResults.push(actionResult);
      
      // Log each action
      await logAction(executionId, action, actionResult);
    }
    
    // 3. Send output
    await sendOutput(agent.output, actionResults);
    
    // 4. Log success
    const duration = Date.now() - startTime;
    await logExecution(executionId, 'success', {
      agentId: agent.id,
      duration,
      results: actionResults
    });
    
    // 5. Update agent stats
    await updateAgentStats(agent.id, {
      totalRuns: agent.totalRuns + 1,
      successfulRuns: agent.successfulRuns + 1,
      lastRun: new Date()
    });
    
  } catch (error) {
    // Log failure
    const duration = Date.now() - startTime;
    await logExecution(executionId, 'failed', {
      agentId: agent.id,
      duration,
      error: error.message
    });
    
    // Update stats
    await updateAgentStats(agent.id, {
      totalRuns: agent.totalRuns + 1,
      failedRuns: agent.failedRuns + 1,
      lastRun: new Date()
    });
    
    // Retry if configured
    if (agent.settings.retryOnFailure && agent.settings.maxRetries > 0) {
      await scheduleRetry(agent, executionId);
    }
    
    // Notify user if configured
    if (agent.settings.notifyOnFailure) {
      await notifyFailure(agent, error);
    }
  }
}

// Execute single action
async function executeAction(action: Action, previousResults: any[]) {
  switch (action.type) {
    case 'generate_prediction':
      return await generateVedicPrediction(action.config);
      
    case 'analyze_video':
      return await analyzeVideo(action.config);
      
    case 'generate_image':
      return await generateImage(action.config);
      
    case 'send_email':
      return await sendEmail(action.config);
      
    case 'send_sms':
      return await sendSMS(action.config);
      
    case 'post_to_slack':
      return await postToSlack(action.config);
      
    // ... many more action types
      
    default:
      throw new Error(`Unknown action type: ${action.type}`);
  }
}
```

```sql
-- Agents table
CREATE TABLE automated_agents (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  
  name TEXT NOT NULL,
  description TEXT,
  
  -- Trigger config
  trigger_type TEXT NOT NULL,
  trigger_config JSONB NOT NULL,
  
  -- Actions
  actions JSONB NOT NULL,
  
  -- Output
  output_type TEXT NOT NULL,
  output_config JSONB NOT NULL,
  
  -- Status
  status TEXT DEFAULT 'active',
  
  -- Stats
  total_runs INTEGER DEFAULT 0,
  successful_runs INTEGER DEFAULT 0,
  failed_runs INTEGER DEFAULT 0,
  last_run TIMESTAMP,
  next_run TIMESTAMP,
  
  -- Settings
  settings JSONB DEFAULT '{}',
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Agent execution logs
CREATE TABLE agent_executions (
  id UUID PRIMARY KEY,
  agent_id UUID REFERENCES automated_agents(id),
  
  status TEXT NOT NULL,  -- 'started', 'success', 'failed'
  duration INTEGER,  -- milliseconds
  
  -- Results
  results JSONB,
  error TEXT,
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Scheduled jobs
CREATE TABLE agent_schedule (
  id UUID PRIMARY KEY,
  agent_id UUID REFERENCES automated_agents(id),
  
  scheduled_for TIMESTAMP NOT NULL,
  status TEXT DEFAULT 'pending',  -- 'pending', 'completed', 'failed'
  
  created_at TIMESTAMP DEFAULT NOW()
);
```
